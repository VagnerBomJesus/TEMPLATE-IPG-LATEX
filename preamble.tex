% --- Língua ---
\usepackage[portuguese]{babel}

% --- Fontes (escolhidas conforme o compilador) ---
\usepackage{iftex}
\ifPDFTeX
    % pdfLaTeX (por defeito no Overleaf): nao acede as fontes do sistema, por
    % isso usam-se equivalentes livres incluidos no TeX Live.
    \usepackage[T1]{fontenc}
    \usepackage[utf8]{inputenc}
    \usepackage{newtxtext}          % serifada  ~ Times New Roman
    \usepackage{carlito}            % sem serifa ~ Calibri (mesmas metricas)
    \renewcommand{\ttdefault}{pcr}  % monoespacada ~ Courier
\else
    % XeLaTeX / LuaLaTeX: fontes reais do modelo Word oficial da UPG.
    \usepackage{fontspec}
    \setmainfont{Times New Roman}
    % Calibri so existe no Windows. No Overleaf / Linux / macOS usa-se a
    % Carlito (metricamente compativel e incluida no TeX Live).
    % O teste e feito pelo ficheiro e nao com \IfFontExistsTF: procurar pelo
    % nome uma fonte inexistente custa tempo em cada passagem e faz o plano
    % gratuito do Overleaf exceder o tempo de compilacao.
    \IfFileExists{C:/Windows/Fonts/calibri.ttf}{%
        \setsansfont{Calibri}[
            BoldFont       = Calibri Bold,
            ItalicFont     = Calibri Italic,
            BoldItalicFont = Calibri Bold Italic
        ]%
    }{%
        \setsansfont{Carlito}[
            Extension      = .ttf,
            UprightFont    = *-Regular,
            BoldFont       = *-Bold,
            ItalicFont     = *-Italic,
            BoldItalicFont = *-BoldItalic
        ]%
    }
    \setmonofont{Courier New}
\fi

% --- Geometria da página ---
\usepackage[a4paper,top=2.5cm,bottom=2.5cm,inner=3cm,outer=2.5cm]{geometry}

% --- Espaçamento entre linhas ---
\usepackage{setspace}

% --- Headings (titlesec) ---
\usepackage{titlesec}
\titleformat{\chapter}[block]{\sffamily\bfseries\fontsize{18}{22}\selectfont}{\thechapter.}{0.5em}{}
\titlespacing*{\chapter}{0pt}{0pt}{18pt}
\titleformat{\section}{\sffamily\bfseries\fontsize{16}{20}\selectfont}{\thesection}{0.5em}{}
\titlespacing*{\section}{0pt}{18pt}{6pt}
\titleformat{\subsection}{\sffamily\bfseries\fontsize{14}{18}\selectfont}{\thesubsection}{0.5em}{}
\titlespacing*{\subsection}{0pt}{14pt}{4pt}
\titleformat{\subsubsection}{\sffamily\fontsize{12}{15}\selectfont}{\thesubsubsection}{0.5em}{}
\titlespacing*{\subsubsection}{0pt}{12pt}{4pt}
\titleformat{\paragraph}[hang]{\sffamily\fontsize{12}{15}\selectfont}{\theparagraph}{0.5em}{}
\titlespacing*{\paragraph}{0pt}{10pt}{4pt}
